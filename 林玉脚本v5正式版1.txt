local function getPetShopFolder()
    local shared = ReplicatedStorage:FindFirstChild("shared")
    local runtime = shared and shared:FindFirstChild("runtime")
    return (runtime and runtime:FindFirstChild("cPetShopFolder"))
        or ReplicatedStorage:FindFirstChild("cPetShopFolder")
end

local function getPetShopRemote()
    local rEvents = ReplicatedStorage:FindFirstChild("rEvents")
    return (rEvents and rEvents:FindFirstChild("cPetShopRemote"))
        or ReplicatedStorage:FindFirstChild("cPetShopRemote")
end

local function filterAvailableItems(names)
    local available = {}
    local shopFolder = getPetShopFolder()
    for _, name in ipairs(names) do
        if not shopFolder or shopFolder:FindFirstChild(name) then
            available[#available + 1] = name
        end
    end
    return available
end

local function purchaseItem(itemName, showNotification)
    if type(itemName) ~= "string" or itemName == "" then
        if showNotification then notify("请先选择一个选项") end
        return false
    end
    local shopFolder = getPetShopFolder()
    local shopRemote = getPetShopRemote()
    local shopItem = shopFolder and shopFolder:FindFirstChild(itemName)
    if not shopFolder or not shopItem or not shopRemote or not shopRemote:IsA("RemoteFunction") then
        if showNotification then notify("商店当前不可用") end
        return false
    end
    local ok, result = pcall(function()
        return shopRemote:InvokeServer(shopItem)
    end)
    if not ok then
        if showNotification then notify("购买失败") end
    end
    return ok and result == true
end

local function LY2_wc(parent, name, tm)
    tm = tm or 10
    if not parent then return nil end
    local o = parent:FindFirstChild(name)
    if o then return o end
    local endT = tick() + tm
    while tick() < endT do
        o = parent:FindFirstChild(name)
        if o then return o end
        task.wait(0.1)
    end
    return nil
end

local function LY2_fmt(n)
    if n >= 1e18 then return string.format("%.2fqi", n / 1e18)
    elseif n >= 1e15 then return string.format("%.2fqa", n / 1e15)
    elseif n >= 1e12 then return string.format("%.2ft", n / 1e12)
    elseif n >= 1e9 then return string.format("%.2fb", n / 1e9)
    elseif n >= 1e6 then return string.format("%.2fm", n / 1e6)
    elseif n >= 1e3 then return string.format("%.2fk", n / 1e3)
    else return tostring(n) end
end

local function LY2_ffull(n)
    local s = tostring(math.floor(n))
    while true do
        local k
        s, k = s:gsub("^(-?%d+)(%d%d%d)", "%1,%2")
        if k == 0 then break end
    end
    return s
end

local function LY2_setD(el, text)
    if not el then return end
    if not pcall(function() el:SetDesc(text) end) then
        if not pcall(function() el:Set({ Desc = text }) end) then
            pcall(function() el:Set(text) end)
        end
    end
end

end

local function LY2_alive(p)
    return p and p.Character and p.Character:FindFirstChild("HumanoidRootPart")
        and p.Character:FindFirstChild("Humanoid") and p.Character.Humanoid.Health > 0
end

local function LY2_prot(p)
    if not p or not p.Character then return false end
    if p.Character:FindFirstChildOfClass("ForceField") then return true end
    local t = p.Character:GetAttribute("SpawnProtectedUntil")
    return typeof(t) == "number" and workspace:GetServerTimeNow() < t
end

local function LY2_myRoot()
    local c = LP.Character
    local h = c and c:FindFirstChildOfClass("Humanoid")
    local r = c and c:FindFirstChild("HumanoidRootPart")
    if not c or not h or h.Health <= 0 or not r then return nil end
    return c, h, r
end

local function LY2_inL(list, n)
    for _, v in ipairs(list) do
        if v:lower() == n:lower() then return true end
    end
    return false
end

local function LY2_near(t, oy)
    if not LY2_alive(t) or LY2_prot(t) or LY2_prot(LP) then return false end
    local c, h, r = LY2_myRoot()
    local tr = t.Character and t.Character:FindFirstChild("HumanoidRootPart")
    if not r or not tr then return false end
    r.CFrame = tr.CFrame * CFrame.new(0, oy, 0)
    r.AssemblyLinearVelocity = Vector3.zero
    r.AssemblyAngularVelocity = Vector3.zero
    return true
end

local function LY2_punch(t, oy)
    if not LY2_near(t, oy) then return false, false end
    LY2_RunS.Heartbeat:Wait()
    if not LY2_near(t, oy) then return false, false end
    local c = LY2_myRoot()
    local tr = t.Character and t.Character:FindFirstChild("HumanoidRootPart")
    local th = t.Character and t.Character:FindFirstChildOfClass("Humanoid")
    if not c or not tr or not th or LY2_prot(t) then return false, false end
    local hand = c:FindFirstChild("LeftHand") or c:FindFirstChild("RightHand")
    if not hand then return false, false end
    local tool = c:FindFirstChild("Punch") or (LP:FindFirstChild("Backpack") and LP.Backpack:FindFirstChild("Punch"))
    local h = c:FindFirstChildOfClass("Humanoid")
    if tool and tool.Parent ~= c and h then pcall(function() h:EquipTool(tool) end) end
    local before = th.Health
    local okP = pcall(function()
        firetouchinterest(tr, hand, 0)
        firetouchinterest(tr, hand, 1)
        if LY2_musEv then LY2_musEv:FireServer("punch", hand.Name == "LeftHand" and "leftHand" or "rightHand") end
    end)
    if not okP then return false, false end
    for _ = 1, 4 do
        LY2_RunS.Heartbeat:Wait()
        if not LY2_alive(t) or th.Health < before then return true, true end
        if LY2_prot(t) then return false, false end
    end
    return false, true
end

local function LY2_hitLimit(t, lim, keep, oy)
    local hits = 0
    while hits < lim and keep() and LY2_alive(t) and not LY2_prot(t) do
        local ok, tried = LY2_punch(t, oy)
        if not tried then break end
        if ok then hits = hits + 1 else task.wait(0.05) end
    end
end

local function LY2_getStat(p, n, lead)
    if lead then
        local l = p and p:FindFirstChild("leaderstats")
        return l and l:FindFirstChild(n)
    end
    return p and p:FindFirstChild(n)
end

local function LY2_st(p, n, l)

    }, "\n")
end

local function LY2_calcHP(p)
    local d = LY2_getStat(p, "Durability", false)
    if not d then return 0 end
    local m = 1
    local u = p:FindFirstChild("ultimatesFolder")
    if u and u:FindFirstChild("Infernal Health") then
        m = m + 0.15 * (u["Infernal Health"].Value or 0)
    end
    local eq = p:FindFirstChild("equippedPets")
    if eq then
        for _, pv in ipairs(eq:GetChildren()) do
            if pv:IsA("ObjectValue") and pv.Value then
                local n = pv.Value.Name:lower()
                if n:find("mighty") and n:find("monster") then m = m + 0.5 end
                if n:find("small") and n:find("fry") then m = m + 0.25 end
            end
        end
    end
    return d.Value * m
end

local function LY2_calcDMG(p)
    local l = p and p:FindFirstChild("leaderstats")
    local s = l and l:FindFirstChild("Strength")
    if not s then return 0 end
    local b = s.Value * 0.0667
    local m = 1
    local u = p:FindFirstChild("ultimatesFolder")
    if u and u:FindFirstChild("Demon Damage") then
        m = m + 0.1 * (u["Demon Damage"].Value or 0)
    end
    local eq = p:FindFirstChild("equippedPets")
    if eq then
        for _, pv in ipairs(eq:GetChildren()) do
            if pv:IsA("ObjectValue") and pv.Value then
                local n = pv.Value.Name:lower()
                if n:find("wild") and n:find("wizard") then m = m + 0.5 end
                if n:find("chaos") and n:find("sorcerer") then m = m + 0.25 end
            end
        end
    end
    return b * m
end

secZD1:Toggle({
    Title = "刷业报v1",
    Default = AutoStates.AutoKillReport,
    Callback = function(state)
        AutoStates.AutoKillReport = state
        if AutoStates.AutoKillReportLoops then
            for _, c in pairs(AutoStates.AutoKillReportLoops) do
                c:Disconnect()
            end
        end
        AutoStates.AutoKillReportLoops = {}
        if not state then
            if game.Players.LocalPlayer.Character then
                local h = game.Players.LocalPlayer.Character:FindFirstChild("Humanoid")
                if h then h:UnequipTools() end
            end
            return
        end
        local pL = game:GetService("RunService").Heartbeat:Connect(function()
            if not AutoStates.AutoKillReport then
                pL:Disconnect()
                return
            end
            local c = game.Players.LocalPlayer.Character
            if c and c:FindFirstChild("Humanoid") and c.Humanoid.Health > 0 then
                local p = c:FindFirstChild("Punch") or game.Players.LocalPlayer.Backpack:FindFirstChild("Punch")
                if p then
                    p.Parent = c
                    p:Activate()
                end
            end
        end)
        table.insert(AutoStates.AutoKillReportLoops, pL)
        local tL = game:GetService("RunService").Heartbeat:Connect(function()
            if not AutoStates.AutoKillReport then
                tL:Disconnect()
                return
            end
            local c = game.Players.LocalPlayer.Character
            if c and c:FindFirstChild("Humanoid") and c.Humanoid.Health > 0 then
                local p = {}
                for _, pl in ipairs(game.Players:GetPlayers()) do
                    if pl ~= game.Players.LocalPlayer and pl.Character and pl.Character:FindFirstChild("HumanoidRootPart") and pl.Character.Humanoid.Health > 0 then
                        table.insert(p, pl)
                    end
                end
                if #p > 0 then
                    local t = p[math.random(#p)]
                    local r = c:WaitForChild("HumanoidRootPart")
                    r.CFrame = t.Character.HumanoidRootPart.CFrame * CFrame.new(0, 0, 2)
                end
            end
        end)
        table.insert(AutoStates.AutoKillReportLoops, tL)
    end
})

local function refreshOptions()
    local newPetEN = filterAvailableItems(PET_NAMES)
    local newPetCN = {}
    for _, en in ipairs(newPetEN) do
        table.insert(newPetCN, PET_CN[en] or en)
    end
    PetDropdown:Refresh(newPetCN)
    local newAuraEN = filterAvailableItems(AURA_NAMES)
    local newAuraCN = {}
    for _, en in ipairs(newAuraEN) do
        table.insert(newAuraCN, AURA_CN[en] or en)
    end
    AuraDropdown:Refresh(newAuraCN)
    local petStillAvailable = false
    for _, cn in ipairs(newPetCN) do
        if cn == selectedPetCN then
            petStillAvailable = true
            break
        end
    end
    if not petStillAvailable then
        selectedPetCN = newPetCN[1] or ""
        selectedPetEN = newPetEN[1] or ""
        if AutoPetToggle.Value then
            AutoPetToggle:SetValue(false)
        end
    end
    local auraStillAvailable = false
    for _, cn in ipairs(newAuraCN) do
        if cn == selectedAuraCN then
            auraStillAvailable = true
            break
        end
    end
    if not auraStillAvailable then
        selectedAuraCN = newAuraCN[1] or ""
        selectedAuraEN = newAuraEN[1] or ""
        if AutoAuraToggle.Value then
            AutoAuraToggle:SetValue(false)
        end
    end
end

local petShopFolder = getPetShopFolder()
if petShopFolder then
    petShopFolder.ChildAdded:Connect(function()
        task.defer(refreshOptions)
    end)
    petShopFolder.ChildRemoved:Connect(function()
        task.defer(refreshOptions)
    end)
else
    ReplicatedStorage.DescendantAdded:Connect(function(descendant)
        if descendant.Name == "cPetShopFolder" then
            task.defer(refreshOptions)
        end
    end)
end

    Values = LY3_pets,
    Value = LY3_pets[1],
    Callback = function(Value) LY3_selTradePet = Value end
})

secJY1:Toggle({
    Title = "自动交易宠物",
    Value = false,
    Callback = function(Value)
        LY3_tradeOn = Value
        LY3_tradeGen = LY3_tradeGen + 1
        local g = LY3_tradeGen
        if not Value then return end
        task.spawn(function()
            while LY3_tradeOn and LY3_tradeGen == g do
                if LY3_trR and LY3_trT and LY3_selTradePet ~= "无数据" then
                    pcall(function()
                        LY3_trR:FireServer("sendTradeRequest", LY3_trT)
                        task.wait(0.5)
                        local pf = LP:FindFirstChild("petsFolder")
                        local uq = pf and pf:FindFirstChild("Unique")
                        local cnt = 0
                        if uq then
                            for _, pet in ipairs(uq:GetChildren()) do
                                if not LY3_tradeOn then break end
                                if pet.Name == LY3_selTradePet then
                                    LY3_trR:FireServer("offerItem", pet)
                                    cnt = cnt + 1
                                    task.wait(0.01)
                                    if cnt >= 6 then break end
                                end
                            end
                        end
                        task.wait(0.05)
                        if LY3_tradeOn then LY3_trR:FireServer("acceptTrade") end
                    end)
                end
                task.wait(2)
            end
        end)
    end
})
